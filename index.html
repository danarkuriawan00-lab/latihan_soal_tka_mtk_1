Pengaturan Auto-Save Spreadsheet

Hubungkan CBT dengan Google Sheets Anda

Status: Web App URL Tersimpan & Siap Digunakan!
Kode Google Apps Script (Copy & Paste) 
const ACCESS_KEY = 'CBT_OPEN_SUBJECTS';
const SPREADSHEET_ID = '15fTxh8Ei-IFiBuaKhdxW_x57c4kbeXNmImuhjmrdaRE';

function doGet(e) {
  const callback = ((e && e.parameter && e.parameter.callback) || '')
    .replace(/[^a-zA-Z0-9_.$]/g, '');
  const saved = PropertiesService.getScriptProperties().getProperty(ACCESS_KEY);
  const result = JSON.stringify({
    subjects: saved ? JSON.parse(saved) : ['Matematika', 'Bahasa Indonesia'],
    configured: !!saved
  });

  if (callback) {
    return ContentService.createTextOutput(callback + '(' + result + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(result)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents || '{}');

    if (data.action === 'saveAccess') {
      PropertiesService.getScriptProperties().setProperty(
        ACCESS_KEY,
        JSON.stringify(data.subjects || [])
      );
      return ContentService.createTextOutput('Access saved');
    }

    const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheets()[0];
    sheet.appendRow([
      data.waktuSelesai || '', data.nama || '', data.kelas || '', data.noAbsen || '',
      data.tanggalLahir || '', data.mapel || '', data.nilai ?? '', data.ringkasan || ''
    ]);
    return ContentService.createTextOutput('Success');
  } catch (error) {
    return ContentService.createTextOutput('Error: ' + error.message);
  }
}

Deploy sebagai versi baru dengan Execute as: Me dan Who has access: Anyone. Akses mapel disimpan di Script Properties dan tersinkron ke perangkat siswa. Tanpa PIN, siapa pun yang memiliki URL Web App dapat mengubah pengaturan akses.
